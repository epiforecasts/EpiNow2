4/**
5 * Convolution of a vector with a reversed PMF, with a reverse-mode adjoint.
6 *
7 * Implements `convolve_with_rev_pmf(x, y, len)` from
8 * inst/stan/functions/convolve.stan.
9 *
10 * Notation (1-based, as in the Stan code): x has length n, y has length D
11 * and is the reversed PMF, and z is the output of length len, with
12 * n <= len <= n + D - 1. The weight of lag d (d = 0, ..., D - 1) is
13 * w_d = y_{D - d}, and x_s = 0 outside 1, ..., n.
14 *
15 * Forward:
16 * z_t = sum_{d = 0}^{D - 1} w_d x_{t - d}, t = 1, ..., len.
17 * This is evaluated lag by lag, one vector update per lag,
18 * z_{d+1:d+m} += w_d x_{1:m}, m = min(n, len - d),
19 * which is z_{d+1:n} += w_d x_{1:n-d} when len = n. When len > n the
20 * output runs past the end of x and m caps the update at the last input,
21 * so the extra outputs are the tail of the full convolution.
22 *
23 * Reverse: the adjoint of a convolution is the matching correlation.
24 * With zbar the adjoint of z, for each lag d
25 * xbar_{1:m} += w_d zbar_{d+1:d+m},
26 * wbar_d += zbar_{d+1:d+m}^T x_{1:m},
27 * and ybar_{D - d} = wbar_d.
28 *
29 * Both passes run on doubles in one reverse_pass_callback, so the
30 * convolution is one autodiff node rather than one dot_product() node per
31 * output. x and y may each be var or double. The function is put in
32 * stan::math so that Stan-generated code finds it.
33 */

61/**
62 * Forward pass of the convolution on doubles.
63 *
64 * Computes z_{d+1:d+m} += w_d x_{1:m} for each lag d, with w_d = y_{D - d}.
65 *
66 * @param x Input vector of length n.
67 * @param y Reversed PMF of length D.
68 * @param len Output length, n <= len <= n + D - 1.
69 * @return Vector z of length len.
70 */
71inline Eigen::VectorXd convolve_forward(const Eigen::VectorXd& x,
72 const Eigen::VectorXd& y, int len) {
73 const int n = x.size();
74 const int D = y.size();
75 Eigen::VectorXd z = Eigen::VectorXd::Zero(len);
76 for (int d = 0; d < D; ++d) {
77 const int m = std::min(n, len - d);
78 if (m <= 0) {
79 break;
80 }
81 z.segment(d, m) += y(D - 1 - d) * x.head(m);
82 }
83 return z;
84}

86/**
87 * Reverse pass of the convolution on doubles.
88 *
89 * Accumulates xbar_{1:m} += w_d zbar_{d+1:d+m} and
90 * wbar_d += zbar_{d+1:d+m}^T x_{1:m}, writing wbar_d to ybar_{D - d}.
91 *
92 * @param x Input vector of length n.
93 * @param y Reversed PMF of length D.
94 * @param zbar Adjoint of the output, of length len.
95 * @param xbar Adjoint of x, of length n, accumulated into. May be null.
96 * @param ybar Adjoint of y, of length D, accumulated into. May be null.
97 */
98inline void convolve_reverse(const Eigen::VectorXd& x,
99 const Eigen::VectorXd& y,
100 const Eigen::VectorXd& zbar,
101 Eigen::VectorXd* xbar, Eigen::VectorXd* ybar) {
102 const int n = x.size();
103 const int D = y.size();
104 const int len = zbar.size();
105 for (int d = 0; d < D; ++d) {
106 const int m = std::min(n, len - d);
107 if (m <= 0) {
108 break;
109 }
110 if (xbar) {
111 xbar->head(m) += y(D - 1 - d) * zbar.segment(d, m);
112 }
113 if (ybar) {
114 (*ybar)(D - 1 - d) += zbar.segment(d, m).dot(x.head(m));
115 }
116 }
117}
